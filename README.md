用法: a [機密筆記內容/支援多行]   #追加寫入並整檔 GPG 鎖定公鑰加密
      cat 檔案 | a                 #【管道串流】直接吸納字串流加密追加
      a -p [檔案路徑]              #【檔案加密】加密普通檔案或 .gpg 巢狀多層加密
      a -p [檔案] --pass [密碼]    #【無密鑰防窮舉】S2K 65,011,712 輪密碼對稱加密
      a -p [檔案] -u               #【加密直傳】加密後直接上傳至雲端 Gist
      a -x [檔案路徑]              #【解密還原】還原一層加密封裝 (去 .gpg)
      a --migrate-repo [--delete-old] #【抹除歷史】建立全新 Gist 倉庫徹底銷毀舊歷史版本
      a -k 或 a --ledger           #【金鑰歸檔簿】查看檔案與金鑰審計清單
      a -a 或 a --all              #解密並列印今年度主機密文檔
      a -a ./[檔案]                #解密並列印【本地密文檔案】
      a -a [年份/檔名]             #即時檢索並列印雲端 Gist 文檔
      a -s 或 a --sync             #推送今年密文文檔至雲端 Gist (-v 檢視細節)
      a -s [檔案路徑]              #加密推送外部檔案至 Gist
      a -s -u [檔案路徑]           #以明文模式推送純文字檔案至 Gist
      a -l 或 a --list             #檢索雲端 Gist 倉庫全部檔案清單
      a -d --all                   #【批量下載】下載雲端 Gist 倉庫全部檔案至本地
      a -d [檔名] -o [目標路徑]     #【自訂下載】指定檔名/自訂路徑 (例: a -d 1.txt -o ./1.txt)
      a -d --all -o [目標目錄]      #下載全部檔案至自訂本地目錄 (例: a -d --all -o ./)
      a -d [年份/檔名]             #下載雲端密文包裹至本地 (保留 .gpg)
      a -d [檔名] -x               #下載並解密還原原始檔案 (去 .gpg)
      a -r1 或 a -r 1              #刪除【倒數第 1 行】
      a -r1-100 或 a -r 1-100      #刪除【倒數 1 至 100 行】
      a -r [關鍵字]                 #刪除包含該關鍵字的所有行
      a -w 或 a --web              #【網頁管理引擎】啟動 Web 視覺化管理後台
      a -w stop                    #關閉 Web 網頁管理引擎 (切換至待機狀態)
      a -w status                  #查詢 Web 網頁管理引擎運行狀態
      a --set-dir [新路徑]          #修改機密文檔本地存儲目錄
      a --init 或 a -i             #系統配置精靈 (支援 Enter 保留舊值)
